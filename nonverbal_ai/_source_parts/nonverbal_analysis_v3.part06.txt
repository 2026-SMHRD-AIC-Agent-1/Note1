def _test_median():
    assert median([1.0, 2.0, 3.0]) == 2.0

    values_with_outlier = [0.5, 0.5, 0.5, 0.5, 0.05]  # 눈깜빡임 한 프레임(0.05)이 섞인 EAR 상황 가정
    assert median(values_with_outlier) == 0.5
    mean_value = sum(values_with_outlier) / len(values_with_outlier)
    assert mean_value < 0.5  # 평균은 이상치에 끌려 내려감 -> median 채택 이유를 재확인
    print("[PASS] median: 이상치에 강함 (median=0.5, mean=", round(mean_value, 3), ")")

def map_to_db_fields(result: dict) -> dict:
    """
    analyze_video()의 전체 결과(dict)를 받아서, PPT 명세서에 있는

    [입력] result: analyze_video()가 반환한 dict (여러 세부값 포함)
    [출력] DB 필드명 14개로 압축된 dict
        {
            "face_yaw": ..., "gaze_center_ratio": ..., "blink_rate": ...,
            "smile_intensity": ..., "expression_change_count": ...,
            "upper_body_sway": ..., "head_nod_count": ..., "hand_gesture_rate": ...,
            "pause_count": ..., "average_volume": ..., "speaking_speed": ...,
